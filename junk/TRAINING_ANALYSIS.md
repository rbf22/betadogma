```python
def _parse_isoforms(self, isoforms_json):
    """Parse isoform metadata from JSON."""
    isoforms = json.loads(isoforms_json)
    
    # Extract protein sequences
    proteins = [iso['protein_seq'] for iso in isoforms if iso['protein_seq']]
    
    # Extract NMD flags
    nmd_flags = [iso['has_nmd'] for iso in isoforms]
    
    # Extract expression levels
    tpms = [iso['expression_tpm'] for iso in isoforms]
    
    # Extract CDS coordinates
    cds_coords = [(iso['cds_start'], iso['cds_end']) for iso in isoforms]
    
    return {
        'proteins': proteins,
        'nmd_flags': nmd_flags,
        'tpms': tpms,
        'cds_coords': cds_coords,
        'is_canonical': [iso['is_canonical'] for iso in isoforms]
    }
```

### 3. Create Protein Labels
```python
def _create_protein_labels(self, protein_seq, cds_start, cds_end, seq_len):
    """Convert protein sequence to per-position amino acid labels."""
    # Map amino acids to indices (20 AA + stop)
    aa_to_idx = {aa: i for i, aa in enumerate('ACDEFGHIKLMNPQRSTVWY*')}
    
    # Create label array
    labels = np.full(seq_len, -1, dtype=np.int64)  # -1 = ignore
    
    # Fill in CDS region with amino acid labels
    if cds_start >= 0 and cds_end <= seq_len and protein_seq:
        cds_len = cds_end - cds_start
        codon_positions = range(cds_start, cds_end, 3)
        
        for i, pos in enumerate(codon_positions):
            if i < len(protein_seq):
                aa = protein_seq[i]
                if aa in aa_to_idx:
                    # Label all 3 positions of the codon
                    for j in range(3):
                        if pos + j < seq_len:
                            labels[pos + j] = aa_to_idx[aa]
    
    return labels
```

### 4. Variant Augmentation
```python
def _apply_variant_augmentation(self, seq, variants, augment_prob=0.5):
    """Apply random variant from metadata to sequence."""
    if random.random() > augment_prob or not variants:
        return seq, None
    
    # Parse variants
    variants = json.loads(variants)
    if not variants:
        return seq, None
    
    # Select random variant
    variant = random.choice(variants)
    
    # Apply variant to sequence
    mutated_seq = self._apply_variant(seq, variant)
    
    return mutated_seq, variant
```
